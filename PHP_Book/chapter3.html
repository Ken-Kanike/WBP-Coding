1. Class Structure & Access Modifiers

<?php
class Account {
    public string $accountNumber;
    protected float $balance = 0.0;
    private string $pin;

    public function __construct(string $accNo, string $pin, float $initialDeposit = 0.0) {
        $this->accountNumber = $accNo;
        $this->pin = password_hash($pin, PASSWORD_DEFAULT);
        $this->balance = $initialDeposit;
    }

    public function getBalance(): float {
        return $this->balance;
    }
}
?>
